\begin{lexlean}{NpmShard}
\useglossary{lexlean.std.bool@1.2.0}
\useglossary{lexlean.std.nat@1.2.0}
\importmodule{BlobLabel}
\title{Boolean}
\begin{semanticmodule}
structure{
  name: Shard,
  type_parameters: [],
  parameters: [],
  fields: [
    {name: nameLabel, type: Nat},
    {name: contentLabel, type: Nat},
    {name: packedBytes, type: Nat},
    {name: cap, type: Nat},
    {name: holdsBlob, type: Nat}
  ]
}

definition{name: transportNpm, parameters: [], result: Nat, body: 2}

definition{
  name: viaShard,
  parameters: [b : BlobLabel::Blob],
  result: BlobLabel::Blob,
  body: (BlobLabel::withEncoding b (transportNpm) 0)
}

definition{
  name: named,
  parameters: [s : Shard],
  result: Bool,
  body: ((project s nameLabel) =? (project s contentLabel))
}

definition{
  name: publishable,
  parameters: [s : Shard],
  result: Bool,
  body: ((named s) && (not ((project s cap) <? (project s packedBytes))))
}

definition{
  name: servesBlob,
  parameters: [s : Shard],
  result: Bool,
  body: ((named s) && ((project s holdsBlob) =? 1))
}

theorem{
  name: aShardNeverChangesAdmission,
  parameters: [b : BlobLabel::Blob],
  statement: ((BlobLabel::admitted (viaShard b)) = (BlobLabel::admitted b)),
  proof: reflexivity{},
  axioms: []
}

theorem{
  name: aShardNeverChangesWhatTheProverReads,
  parameters: [b : BlobLabel::Blob],
  statement: ((BlobLabel::proverInput (viaShard b)) = (BlobLabel::proverInput b)),
  proof: reflexivity{},
  axioms: []
}

definition{
  name: breaksCap,
  parameters: [s : Shard],
  result: Bool,
  body: ((publishable s) && ((project s cap) <? (project s packedBytes)))
}

theorem{
  name: noShardOverItsCapIsPublished,
  parameters: [s : Shard],
  statement: ((breaksCap s) = false),
  proof: linear_arithmetic{definitions: [breaksCap, named, publishable]},
  axioms: [Quot.sound, propext]
}

definition{
  name: breaksNaming,
  parameters: [s : Shard],
  result: Bool,
  body: ((servesBlob s) && (not ((project s nameLabel) =? (project s contentLabel))))
}

theorem{
  name: aShardServesOnlyUnderTheLabelOfItsContent,
  parameters: [s : Shard],
  statement: ((breaksNaming s) = false),
  proof: linear_arithmetic{definitions: [breaksNaming, named, servesBlob]},
  axioms: [Quot.sound, propext]
}
\end{semanticmodule}
\end{lexlean}
