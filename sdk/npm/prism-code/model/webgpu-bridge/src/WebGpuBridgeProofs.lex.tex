definition{
  name: breaksAdapter,
  parameters: [p : WebGpuBridge::Plan],
  result: Bool,
  body: ((WebGpuBridge::planAdmitted p) && (not ((project p adapter) =? 1)))
}

theorem{
  name: aKernelIsDispatchedOnlyOnAnAdapter,
  parameters: [p : WebGpuBridge::Plan],
  statement: ((breaksAdapter p) = false),
  proof: linear_arithmetic{definitions: [WebGpuBridge::buffersFit, WebGpuBridge::dispatchCovers, WebGpuBridge::kernelBound, WebGpuBridge::planAdmitted, breaksAdapter]},
  axioms: [Quot.sound, propext]
}
