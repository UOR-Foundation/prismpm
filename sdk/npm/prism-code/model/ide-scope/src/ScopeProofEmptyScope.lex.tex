\begin{lexlean}{ScopeProofEmptyScope}
\useglossary{lexlean.std.bool@1.2.0}
\useglossary{lexlean.std.nat@1.2.0}
\importmodule{Scope}
\title{Boolean}
\begin{semanticmodule}
definition{
  name: breaksEmptyScope,
  parameters: [r : Scope::Request],
  result: Bool,
  body: (((Scope::judge r) =? (Scope::answerInside)) && ((project r scopeDepth) =? 0))
}

theorem{
  name: neverEmptyScope,
  parameters: [r : Scope::Request],
  statement: ((breaksEmptyScope r) = false),
  proof: linear_arithmetic{
    definitions: [
      Scope::answerInside,
      Scope::answerOutside,
      Scope::firstName,
      Scope::inside,
      Scope::judge,
      Scope::prefixMatches,
      Scope::scopeNamed,
      Scope::scopeSlots,
      Scope::targetNamed,
      Scope::targetSlots,
      Scope::wellFormed,
      breaksEmptyScope
    ]
  },
  axioms: [Quot.sound, propext]
}
\end{semanticmodule}
\end{lexlean}
